% ---------------------------------------------------------------------- glyphs
% An underground belt: a hood over the half of the tile that faces the tunnel. The mouth
% (wide end) faces the surface belt; an entrance's tunnel runs ahead, an exit's behind.
\tikzset{
	bm entrance/.pic={
		\bm@set{#1}
		\begin{scope}[rotate=\bm@dir]
			\path[bm hood entrance] (-0.06,0.31) -- (0.42,0.19) -- (0.42,-0.19) -- (-0.06,-0.31) -- cycle;
			\path[bm hood mark] (0.13,0.075) -- (0.215,0) -- (0.13,-0.075);
			\ifx\bm@trim\pgfutil@empty\else\path[bm trim bar=bmt-\bm@trim] (-0.06,0.31) -- (-0.06,-0.31);\fi
		\end{scope}
	},
	bm exit/.pic={
		\bm@set{#1}
		\begin{scope}[rotate=\bm@dir]
			\path[bm hood exit] (0.06,0.31) -- (-0.42,0.19) -- (-0.42,-0.19) -- (0.06,-0.31) -- cycle;
			\path[bm hood mark exit] (-0.225,0.075) -- (-0.14,0) -- (-0.225,-0.075);
			\ifx\bm@trim\pgfutil@empty\else\path[bm trim bar=bmt-\bm@trim] (0.06,0.31) -- (0.06,-0.31);\fi
		\end{scope}
	},
	% A splitter, placed at the middle of its two tiles; left is the tile on the left of
	% somebody travelling in its direction. Marks are placed by coordinates in the rotated
	% frame (a shift= would not turn with it).
	bm splitter/.pic={
		\bm@set{#1}
		\begin{scope}[rotate=\bm@dir]
			\ifx\bm@trim\pgfutil@empty
				\path[bm splitter body] (-0.34,-0.84) rectangle (0.34,0.84);
			\else
				\path[bm splitter body, bm trim=bmt-\bm@trim] (-0.34,-0.84) rectangle (0.34,0.84);
			\fi
			% numbers=true: the splitter's number in place of its arrow (text stays upright).
			\ifx\bm@number\pgfutil@empty
				\path[bm splitter arrow] (0.16,0) -- (-0.1,0.13) -- (-0.1,-0.13) -- cycle;
			\else
				\node[bm splitter number] at (0,0) {\bm@number};
			\fi
			\ifx\bm@in\pgfutil@empty\else
				\edef\bm@s{\bm@side\bm@in}\path[bm prio] (-0.30,{\bm@s+0.17}) -- (-0.30,{\bm@s-0.17}) -- (-0.04,\bm@s) -- cycle;
			\fi
			\ifx\bm@filter\pgfutil@empty
				\ifx\bm@out\pgfutil@empty\else
					\edef\bm@s{\bm@side\bm@out}\path[bm prio] (0.04,{\bm@s+0.17}) -- (0.04,{\bm@s-0.17}) -- (0.30,\bm@s) -- cycle;
				\fi
			\else
				% The filtered item leaves on the output-priority side, left when none is set.
				\ifx\bm@out\pgfutil@empty\def\bm@out{left}\fi
				\edef\bm@s{\bm@side\bm@out}\path[bm filter=\bm@fcolor] (0.135,{\bm@s+0.17}) -- (0.305,\bm@s) -- (0.135,{\bm@s-0.17}) -- (-0.035,\bm@s) -- cycle;
				% filter icon={...}: the item's icon over the diamond, upright, on a light plate.
				\ifx\bm@ficon\pgfutil@empty\else
					\node[bm filter icon] at (0.135,\bm@s) {\bm@ficon};
				\fi
			\fi
		\end{scope}
	},
	bm head/.pic={
		\bm@set{#1}
		\path[bm head, rotate=\bm@dir] (0.09,0) -- (-0.055,0.08) -- (-0.055,-0.08) -- cycle;
	},
}
